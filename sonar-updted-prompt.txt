=================================================
CORE ENTITIES & DATA MODEL
=================================================

Product:
- id (Integer PK)
- sku (String Unique, e.g., SKU-GRO-0001)
- name (String)
- category (Enum: grocery, electronics, clothing, household, personal_care)
- unit_price (Float)
- cost_price (Float)
- unit_of_measure (String: pieces, kg, litre, box)
- reorder_point (Integer)
- reorder_quantity (Integer)
- supplier_id (FK -> Supplier)
- created_at (DateTime)

StockLevel:
- id (Integer PK)
- product_id (FK -> Product, Unique)
- quantity_on_hand (Integer)
- quantity_reserved (Integer)
- quantity_available (Property: max(0, quantity_on_hand - quantity_reserved))
- last_updated (DateTime)

StockMovement:
- id (Integer PK)
- product_id (FK -> Product)
- movement_type (Enum: receipt, sale, adjustment, transfer, return)
- quantity (Integer, positive for IN, negative for OUT)
- reference_number (String, e.g., PO-2026-0001 or SALE-001)
- notes (String)
- recorded_at (DateTime)
- recorded_by (String)

PurchaseOrder:
- id (Integer PK)
- po_number (String Unique, e.g., PO-2026-0001)
- supplier_id (FK -> Supplier)
- status (Enum: draft, submitted, acknowledged, received, cancelled)
- total_amount (Float)
- order_date (Date)
- expected_delivery (Date)
- received_date (Date)
- created_at (DateTime)

POItem:
- id (Integer PK)
- po_id (FK -> PurchaseOrder)
- product_id (FK -> Product)
- quantity_ordered (Integer)
- unit_cost (Float)
- quantity_received (Integer)

Supplier:
- id (Integer PK)
- name (String)
- supplier_code (String Unique, e.g., SUP-0001)
- contact_email (String)
- payment_terms_days (Integer)
- lead_time_days (Integer)
- is_active (Boolean)

StockAlert:
- id (Integer PK)
- product_id (FK -> Product)
- alert_type (String: low_stock, out_of_stock)
- message (String)
- is_resolved (Boolean)
- triggered_at (DateTime)

User:
- id (Integer PK)
- email (String Unique)
- hashed_password (String)
- full_name (String)
- role (String: manager, analyst, procurement, warehouse)
- is_active (Boolean)
